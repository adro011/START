# ── MOTION / TIMING GLOBALS ────────────────────────────────────────────
# Shared durations, easing curves and stagger steps. Every animation
# references these, so changing one value retimes the whole site.
#
# KEY                MEANING
# duration.instant   micro feedback (hover tint)
# duration.fast      small UI moves (buttons, links)
# duration.base      default transition
# duration.slow      panels, cards, images
# duration.hero      opening/entrance animations
# ease.standard      default CSS easing
# ease.enter         things appearing
# ease.exit          things leaving
# ease.spring        playful overshoot
# gsap.ease          easing name used by GSAP scripts
# stagger.tight/base/section   delay between grouped items
# reducedMotion      respect | ignore  (accessibility setting)

duration.instant = 100ms
duration.fast = 200ms
duration.base = 350ms
duration.slow = 600ms
duration.hero = 900ms

ease.standard = cubic-bezier(0.22, 1, 0.36, 1)
ease.enter = cubic-bezier(0.22, 1, 0.36, 1)
ease.exit = cubic-bezier(0.4, 0, 1, 1)
ease.spring = cubic-bezier(0.34, 1.56, 0.64, 1)
gsap.ease = power3.out

stagger.tight = 0.04
stagger.base = 0.08
stagger.section = 0.16

reducedMotion = respect
